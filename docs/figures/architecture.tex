}

  \node[dots] at (\xCdots,\y) {$\cdots$};
  \node[dots] at (\xIdots,\y) {$\cdots$};

  \draw[e] (cm1\c) -- (cm2\c);

\foreach \c/\j in {a/1, b/2, g/7} {
