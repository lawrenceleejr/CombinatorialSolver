% a row of tokens:  \tokrow{x}{y}{fill}{name}{lab1}{lab2}{lab3}{lab7}
% draws  [1][2][3] ... [7]  with identical boxes
\newcommand{\tokrow}[8]{%
  \foreach \k/\lab in {0/#5,1/#6,2/#7} {
    \node[tok, fill=#3] (#4\k) at ({#1+\k*\pitch},#2) {};
    \node[lbl, font=\scriptsize, below=1pt of #4\k] {\lab};
  }
  \node[text=quiet] at ({#1+3*\pitch+0.05},#2) {$\cdots$};
  \node[tok, fill=#3] (#4n) at ({#1+4.25*\pitch},#2) {};
  \node[lbl, font=\scriptsize, below=1pt of #4n] {#8};
}
